// Estado global da aplicação
const AppState = {
    currentSlide: 0,
    isEditing: false,
    zoom: 0.6,
    slides: [],
    
    // Inicializa dados dos slides
    init() {
        this.slides = [
            { ...window.slideComponents.slide1.defaultData },
            { ...window.slideComponents.slide2.defaultData },
            { ...window.slideComponents.slide3.defaultData },
            { ...window.slideComponents.slide4.defaultData },
            { ...window.slideComponents.slide5.defaultData },
            { ...window.slideComponents.slide6.defaultData },
            { ...window.slideComponents.slide7.defaultData },
            { ...window.slideComponents.slide8.defaultData }
        ];
        
        // Deep copy para arrays aninhados
        this.slides[2].points = [...window.slideComponents.slide3.defaultData.points];
        this.slides[3].items = window.slideComponents.slide4.defaultData.items.map(i => ({...i}));
    },
    
    // Atualiza dado específico
    update(slideIndex, key, value) {
        if (this.slides[slideIndex]) {
            this.slides[slideIndex][key] = value;
        }
    },
    
    // Duplica slide
    duplicate(index) {
        const toDuplicate = this.slides[index];
        const copy = JSON.parse(JSON.stringify(toDuplicate));
        this.slides.splice(index + 1, 0, copy);
        return this.slides.length;
    },
    
    // Reset slide para default
    reset(index) {
        const component = window.slideComponents[`slide${index + 1}`];
        if (component) {
            this.slides[index] = JSON.parse(JSON.stringify(component.defaultData));
        }
    },
    
    // Exporta para JSON
    exportJSON() {
        return JSON.stringify(this.slides, null, 2);
    },
    
    // Importa de JSON
    importJSON(json) {
        try {
            const data = JSON.parse(json);
            if (Array.isArray(data) && data.length === 8) {
                this.slides = data;
                return true;
            }
            return false;
        } catch(e) {
            return false;
        }
    }
};
